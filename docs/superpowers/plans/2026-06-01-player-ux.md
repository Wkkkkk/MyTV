# Player UX + Mobile/TV UI Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a channel info bar (logo, name, category, position), keyboard shortcuts (space/arrows/F), fullscreen, and responsive CSS polish to the MyTV player — no new endpoints, no schema changes.

**Architecture:** Enrich `TuneResponse` in `player.rs` with channel metadata; add `channel_id` to `ChannelRow` and embed `window.epgChannels` JSON in the EPG partial; add a `#player-info` bar rendered client-side from the enriched response; add a single `keydown` handler for all shortcuts.

**Tech Stack:** Rust/Axum, Askama 0.12 templates, vanilla JS, serde_json 1 (already in Cargo.toml)

```javascript
let currentChannel = null;

function hashColor(str) {
  var h = 0;
  for (var i = 0; i < str.length; i++) {
    h = (str.charCodeAt(i) + ((h << 5) - h)) | 0;
  }
  return 'hsl(' + (Math.abs(h) % 360) + ',35%,28%)';
}

function channelIcon(category) {
  var c = (category || '').toLowerCase();
  if (c.indexOf('news') >= 0) return '📰';
  if (c.indexOf('sport') >= 0) return '⚽';
  if (c.indexOf('movie') >= 0 || c.indexOf('film') >= 0 || c.indexOf('cinema') >= 0) return '🎬';
  if (c.indexOf('music') >= 0) return '🎵';
  if (c.indexOf('kid') >= 0 || c.indexOf('child') >= 0) return '🧒';
  if (c.indexOf('documentary') >= 0 || c.indexOf('docu') >= 0) return '🎥';
  if (c.indexOf('entertainment') >= 0) return '🎭';
  if (c.indexOf('cooking') >= 0 || c.indexOf('food') >= 0) return '🍳';
  if (c.indexOf('travel') >= 0) return '✈️';
  if (c.indexOf('science') >= 0 || c.indexOf('tech') >= 0) return '🔬';
  return '📺';
}

function renderInfoBar(channel) {
  var bar = document.getElementById('player-info');
  if (!bar || !channel) return;

  var logoEl = bar.querySelector('.pi-logo');
  if (channel.logo_url) {
    logoEl.style.background = 'transparent';
    logoEl.innerHTML = '<img src="' + channel.logo_url.replace(/['"]/g, '') + '" alt="">';
  } else {
    logoEl.style.background = hashColor(channel.name || '');
    logoEl.innerHTML = '<span>' + escHtml((channel.name || '?')[0].toUpperCase()) + '</span>';
  }

  bar.querySelector('.pi-name').textContent = channel.name || '';
  bar.querySelector('.pi-cat-icon').textContent = channelIcon(channel.category);
  bar.querySelector('.pi-cat-text').textContent = channel.category ? ' ' + channel.category : '';

  var liveEl = bar.querySelector('.pi-live');
  liveEl.style.display = channel.channel_type === 'live' ? '' : 'none';

  var posEl = bar.querySelector('.pi-pos');
  var channels = window.epgChannels || [];
  var idx = -1;
  for (var i = 0; i < channels.length; i++) {
    if (channels[i].id === channel.channel_id) { idx = i; break; }
  }
  if (idx >= 0) {
    posEl.textContent = 'Ch ' + (idx + 1) + ' / ' + channels.length;
    posEl.style.display = '';
  } else {
    posEl.style.display = 'none';
  }

  bar.style.display = 'flex';
}
```

- [ ] **Step 2: Update the `tune()` function to store channel data**

Find the existing `function tune(channelId) {` and update the `.then()` callback that calls `_loadSource` so it also stores `currentChannel` and calls `renderInfoBar`:

```javascript
function tune(channelId) {
  currentChannelId = channelId;
  currentUrl = null;
  hidePlayerError();
  document.getElementById('player-panel').style.display = 'block';
  fetch('/channel/' + channelId + '/tune')
    .then(function(r) {
      if (!r.ok) throw new Error('tune failed: ' + r.status);
      return r.json();
    })
    .then(function(d) {
      currentChannel = Object.assign({ channel_id: channelId }, d);
      renderInfoBar(currentChannel);
      _loadSource(d.url, d.start_offset_secs);
    })
    .catch(function(err) {
      if (typeof debugLog === 'function') debugLog('error', 'tune: ' + err);
      console.error('tune error:', err);
      showPlayerError();
    });
}
```

`Object.assign({ channel_id: channelId }, d)` merges the channelId (already known from the function argument) with all fields from the server response (`name`, `logo_url`, `category`, `channel_type`, `url`, `start_offset_secs`).

```javascript
document.addEventListener('keydown', function(e) {
  var tag = (e.target.tagName || '').toLowerCase();
  if (tag === 'input' || tag === 'textarea' || tag === 'select') return;

  if (e.key === ' ') {
    e.preventDefault();
    if (!video) return;
    if (video.paused) { video.play().catch(function(){}); }
    else { video.pause(); }
    return;
  }

  if (e.key === 'f' || e.key === 'F') {
    if (!video) return;
    if (document.fullscreenElement) {
      document.exitFullscreen();
    } else {
      video.requestFullscreen().catch(function(){});
    }
    return;
  }

  if (e.key === 'ArrowUp' || e.key === 'ArrowDown') {
    e.preventDefault();
    var channels = window.epgChannels || [];
    if (!channels.length || !currentChannelId) return;
    var idx = -1;
    for (var i = 0; i < channels.length; i++) {
      if (channels[i].id === currentChannelId) { idx = i; break; }
    }
    if (idx < 0) idx = 0;
    var next = e.key === 'ArrowDown'
      ? (idx + 1) % channels.length
      : (idx - 1 + channels.length) % channels.length;
    tune(channels[next].id);
    return;
  }

  if (e.key === 'ArrowLeft' || e.key === 'ArrowRight') {
    if (!currentChannel || currentChannel.channel_type !== 'vod_loop') return;
    if (!video) return;
    e.preventDefault();
    video.currentTime += e.key === 'ArrowRight' ? 10 : -10;
    return;
  }
});
```
